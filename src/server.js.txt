// server.js

// Import required modules
const express = require('express');
const path = require('path');

// Create an Express application
const app = express();

// Define the port the server will run on.
// It will use the port from the environment variables (for hosting platforms)
// or default to 3000 for local development.
const PORT = process.env.PORT || 3000;

// Serve static files (HTML, CSS, JS) from the 'public' directory
app.use(express.static(path.join(__dirname, 'public')));

// A catch-all route to send the main index.html file for any request
// that doesn't match a static file. This is useful for single-page applications.
app.get('*', (req, res) => {
    res.sendFile(path.join(__dirname, 'public', 'index.html'));
});

// Start the server and listen for incoming connections on the specified port
app.listen(PORT, () => {
    console.log(`Server is running on http://localhost:${PORT}`);
});
